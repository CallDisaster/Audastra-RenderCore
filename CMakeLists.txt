cmake_minimum_required(VERSION 3.24)
project(War3RenderCore VERSION 0.1.16 LANGUAGES CXX)
option(WAR3_RENDER_CORE_BUILD_TESTS "构建可移植 CPU 回归" ${PROJECT_IS_TOP_LEVEL})
option(WAR3_RENDER_CORE_BUILD_TOOLS "构建纯数据黄金夹具工具" ${PROJECT_IS_TOP_LEVEL})
set(WAR3_RENDER_CORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(war3-render-contract INTERFACE)
add_library(War3RenderCore::Contract ALIAS war3-render-contract)
target_include_directories(war3-render-contract INTERFACE "${WAR3_RENDER_CORE_ROOT}/include")
target_compile_features(war3-render-contract INTERFACE cxx_std_20)
function(war3_render_core_options target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /fp:precise)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()
endfunction()
add_library(war3-mdx STATIC src/Render/Assets/MdxModel.cpp src/Render/Assets/MdxPose.cpp)
add_library(War3RenderCore::MDX ALIAS war3-mdx)
target_link_libraries(war3-mdx PUBLIC War3RenderCore::Contract)
war3_render_core_options(war3-mdx)
add_subdirectory(src/RenderCore)
# 原插件输入测试的标准库闭包；不包含设备、窗口、GPU 后端或安装逻辑。
add_library(war3-render-skin-input STATIC src/Render/Scene/ModelSkinInput.cpp)
add_library(War3RenderCore::SkinInput ALIAS war3-render-skin-input)
target_link_libraries(war3-render-skin-input PUBLIC War3RenderCore::Contract)
war3_render_core_options(war3-render-skin-input)
add_library(war3-mdx-scan-data STATIC src/Render/Tools/MdxScan/ScanRecord.cpp src/Render/Tools/MdxScan/ScanDetail.cpp)
add_library(War3RenderCore::ScanData ALIAS war3-mdx-scan-data)
target_link_libraries(war3-mdx-scan-data PUBLIC War3RenderCore::CPU)
war3_render_core_options(war3-mdx-scan-data)
if(WAR3_RENDER_CORE_BUILD_TESTS)
    enable_testing()
    include(src/RenderCore/TestTargets.cmake)
endif()
if(WAR3_RENDER_CORE_BUILD_TOOLS)
    add_executable(war3-render-core-footman-golden tools/FootmanGolden.cpp)
    target_link_libraries(war3-render-core-footman-golden PRIVATE War3RenderCore::CPU)
    war3_render_core_options(war3-render-core-footman-golden)
endif()
